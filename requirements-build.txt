# Build-only dependencies for PyInstaller and Nuitka.
#
# These are NOT runtime dependencies — they are needed to produce
# distributable binaries. CI workers that only run tests do not
# need this file. The build_windows.sh / build_windows_docker.py
# and build_wine_local.py scripts install from this file in their
# builder images.
#
# PyInstaller is the current default builder. Nuitka is being
# introduced alongside it (Phase 5 of the Qt6/PySide6 migration
# plan) and will become the default after one shipping release
# validates it.

# PyInstaller (current default — being phased out)
pyinstaller>=6.0.0
pyinstaller-hooks-contrib>=2024.0

# Nuitka (Phase 5 target)
nuitka>=2.4

# Nuitka transitive deps
ordered-set>=4.1

# Linux build tools (Nuitka on Linux uses patchelf to fix rpath;
# PyInstaller on Linux uses it to strip debug symbols)
patchelf>=0.17

# pip and must be supplied by the builder image (see
